\noindent
Symbols used throughout the thesis are listed below, grouped by theme.
SI units follow the conventions of \texttt{siunitx}; dimensionless
quantities are marked $[-]$. Acronyms (e.g.\ PUBM, RUBM, FIIN, RDSO,
STL) are listed in the \emph{List of Abbreviations}.

\subsection*{\color{gsvblue}Apparatus and loading}

{\small
\renewcommand{\arraystretch}{1.12}
\rowcolors{2}{rowbg}{white}
\begin{tabular}{@{}p{0.18\textwidth}p{0.62\textwidth}p{0.14\textwidth}@{}}
\toprule
\rowcolor{tablehdr}
{\color{white}\textbf{Symbol}} & {\color{white}\textbf{Meaning}} & {\color{white}\textbf{Units}}\\
\midrule
$m$            & Hammer mass                                         & \si{\kilogram} \\
$h$            & Hammer drop height (above pad)                       & \si{\milli\metre} or \si{\metre} \\

$\Ke$          & Kinetic energy of hammer at impact, $mgh$            & \si{\joule} \\
$D$            & Mould internal diameter                              & \si{\milli\metre} \\
$\Ap$          & Cross-sectional area of pad/specimen                 & \si{\milli\metre\squared} \\
$F$            & Peak hammer force (in-line load cell)                & \si{\kilo\newton} \\
$\sigma$       & Nominal contact stress at specimen surface, $F/\Ap$  & \si{\kilo\pascal} \\
$N$            & Cumulative number of impact blows                    & $[-]$ \\
\bottomrule
\end{tabular}}

\subsection*{\color{gsvblue}Per-particle geometry (3D mesh-derived)}

{\small
\renewcommand{\arraystretch}{1.12}
\rowcolors{2}{rowbg}{white}
\begin{tabular}{@{}p{0.18\textwidth}p{0.62\textwidth}p{0.14\textwidth}@{}}
\toprule
\rowcolor{tablehdr}
{\color{white}\textbf{Symbol}} & {\color{white}\textbf{Meaning}} & {\color{white}\textbf{Units}}\\
\midrule
$V$            & Particle volume                                     & \si{\milli\metre\cubed} \\
$A$            & Particle surface area                               & \si{\milli\metre\squared} \\
$\Vch$         & Convex-hull volume                                  & \si{\milli\metre\cubed} \\
$\Ach$         & Convex-hull surface area                            & \si{\milli\metre\squared} \\
$\Deq$         & Equivalent spherical diameter,
                 $\left(6V/\pi\right)^{1/3}$                          & \si{\milli\metre} \\
$L, I, S$      & Long, intermediate, and short bounding-box dimensions
                 ($L\ge I\ge S$)                                      & \si{\milli\metre} \\
$a,b,c$        & Semi-axes of best-fit ellipsoid ($a\ge b\ge c$)     & \si{\milli\metre} \\
$\Psi$         & Sphericity = $\pi^{1/3}(6V)^{2/3}/A$
                 \citep{Wadell1932}                                   & $[-]$ \\
$\kappa_{s}$   & Sphericity ratio = $\Deq/L$                          & $[-]$ \\
$C$            & Convexity = $V/\Vch$                                 & $[-]$ \\
$T$            & Surface-texture index = $A/\Ach$                    & $[-]$ \\
$E$            & Elongation = $I/L$                                  & $[-]$ \\
$\Ab/\Ap$      & Flatness = $S/I$ (alternative form $(L+I)/(2S)$)    & $[-]$ \\
$R$            & Roundness descriptor (scanner-specific)             & $[-]$ \\
$\dmax$        & Maximum gradation particle size                      & \si{\milli\metre} \\
\bottomrule
\end{tabular}}

\subsection*{\color{gsvblue}Paired-change operators}

{\small
\renewcommand{\arraystretch}{1.12}
\rowcolors{2}{rowbg}{white}
\begin{tabular}{@{}p{0.18\textwidth}p{0.62\textwidth}p{0.14\textwidth}@{}}
\toprule
\rowcolor{tablehdr}
{\color{white}\textbf{Symbol}} & {\color{white}\textbf{Meaning}} & {\color{white}\textbf{Units}}\\
\midrule
$\Delta X$     & Change in descriptor $X$:
                 $X_{\text{after}} - X_{\text{before}}$                & units of $X$ \\
$\%\Delta X$   & Percentage change in $X$:
                 $\Delta X / X_{\text{before}} \times 100$             & \si{\percent} \\
$\overline{\Delta X}$ & Population mean of paired changes              & units of $X$ \\
$\widetilde{\Delta X}$ & Population median of paired changes           & units of $X$ \\
\bottomrule
\end{tabular}}
